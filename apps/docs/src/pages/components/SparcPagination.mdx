import {
  Pagination,
  PaginationFull,
  PaginationPageButton,
  PaginationRowsPerPage,
  PaginationSummary,
} from '@sparc-ui';
import { useState } from 'react';

export function BasicDemo() {
  const [page, setPage] = useState(1);
  return <Pagination page={page} pageCount={10} onPageChange={setPage} />;
}

export function EllipsisStartDemo() {
  const [page, setPage] = useState(1);
  return <Pagination page={page} pageCount={23} onPageChange={setPage} />;
}

export function EllipsisMiddleDemo() {
  const [page, setPage] = useState(12);
  return <Pagination page={page} pageCount={23} onPageChange={setPage} />;
}

export function RowsPerPageDemo() {
  const [value, setValue] = useState(25);
  return <PaginationRowsPerPage value={value} onChange={setValue} />;
}

export function FullStartDemo() {
  const [page, setPage] = useState(1);
  const [pageSize, setPageSize] = useState(25);
  return (
    <PaginationFull
      layout="summary-start"
      page={page}
      pageCount={1}
      onPageChange={setPage}
      pageSize={pageSize}
      onPageSizeChange={setPageSize}
      total={25}
    />
  );
}

export function FullEndDemo() {
  const [page, setPage] = useState(1);
  const [pageSize, setPageSize] = useState(25);
  return (
    <PaginationFull
      layout="summary-end"
      page={page}
      pageCount={12}
      onPageChange={setPage}
      pageSize={pageSize}
      onPageSizeChange={setPageSize}
      total={300}
    />
  );
}

## With ellipsis

With `pageCount` larger than the edge blocks, pages collapse to `1 2 3 … 21 22 23` near the start (the exact Figma pattern, `siblingCount=1`) and `1 … 11 12 13 … 23` mid-range. The ellipsis button jumps `2*siblingCount+1` pages.

## Number page states

The isolated Figma cells were ambiguous, so the mapping is a documented decision: default = white, hover = `#F5F5F5`, current = white with a black/10 border (`aria-current="page"`).

## Props

### Pagination

| Prop | Type | Default | Description |
|---|---|---|---|
| `page` | `number` | — | Controlled 1-based current page |
| `pageCount` | `number` | — | Total number of pages |
| `onPageChange` | `(page: number) => void` | — | Called with the requested page |
| `siblingCount` | `number` | `1` | Pages each side of current; sets edge block size and ellipsis jump |
| `previousLabel` / `nextLabel` | `string` | `"Previous"` / `"Next"` | Nav button labels |
| `className` | `string` | — | Extra classes on the `<nav>` |

### PaginationRowsPerPage

| Prop | Type | Default | Description |
|---|---|---|---|
| `value` | `number` | — | Controlled selected value |
| `options` | `number[]` | `[10, 25, 50, 100]` | Selectable values |
| `onChange` | `(value: number) => void` | — | Called with the chosen value |
| `label` | `string` | `"Rows per page"` | Leading label text |
| `open` | `boolean` | — | Force the panel open (docs) |
| `disabled` | `boolean` | `false` | Disabled state |

### PaginationSummary

| Prop | Type | Default | Description |
|---|---|---|---|
| `page` | `number` | — | Current page |
| `pageSize` | `number` | — | Rows per page |
| `total` | `number` | — | Total rows |

### PaginationFull

| Prop | Type | Default | Description |
|---|---|---|---|
| `layout` | `"summary-start" \| "summary-end"` | `"summary-start"` | Figma arrangement A / B |
| `pageSize` | `number` | — | Rows-per-page value |
| `onPageSizeChange` | `(value: number) => void` | — | Rows-per-page change handler |
| `pageSizeOptions` | `number[]` | `[10, 25, 50, 100]` | Rows-per-page options |
| `total` | `number` | — | Total rows (summary) |
| `rowsPerPageLabel` | `string` | `"Rows per page"` | Rows-per-page label |

All `Pagination` props are forwarded to the inner pagination.
