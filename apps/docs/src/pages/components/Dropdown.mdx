export function VariantInteractiveExample() {
  const [openDefault, setOpenDefault] = useState(false);
  const [openTags, setOpenTags] = useState(false);
  const [selected, setSelected] = useState(null);
  const [tags, setTags] = useState([
    { value: 'a', label: 'LABEL' },
    { value: 'b', label: 'LABEL' },
  ]);
  return (
    <div className="grid grid-cols-2 gap-6">
      <Dropdown
        variant="default"
        label="Default"
        placeholder="Select"
        hint="This is a hint text to help user."
        open={openDefault}
        selectedLabel={selected ?? undefined}
        onClick={() => setOpenDefault((o) => !o)}
        popup={
          <DropdownMenu>
            <DropdownItem
              selected={selected === 'Option A'}
              showSelectedMark
              onClick={() => { setSelected('Option A'); setOpenDefault(false); }}
            >
              Option A
            </DropdownItem>
            <DropdownItem
              selected={selected === 'Option B'}
              showSelectedMark
              onClick={() => { setSelected('Option B'); setOpenDefault(false); }}
            >
              Option B
            </DropdownItem>
          </DropdownMenu>
        }
      />
      <Dropdown
        variant="tags"
        label="Tags"
        placeholder="Select"
        hint="This is a hint text to help user."
        tags={tags}
        onRemoveTag={(v) => setTags((arr) => arr.filter((t) => t.value !== v))}
        open={openTags}
        onClick={() => setOpenTags((o) => !o)}
        popup={
          <DropdownMenu>
            <DropdownItem
              leftAdornment={<Checkbox checked={tags.some((t) => t.value === 'a')} onCheckedChange={() => {}} />}
              onClick={() =>
                setTags((arr) =>
                  arr.some((t) => t.value === 'a')
                    ? arr.filter((t) => t.value !== 'a')
                    : [...arr, { value: 'a', label: 'LABEL' }],
                )
              }
            >
              Label A
            </DropdownItem>
            <DropdownItem
              leftAdornment={<Checkbox checked={tags.some((t) => t.value === 'b')} onCheckedChange={() => {}} />}
              onClick={() =>
                setTags((arr) =>
                  arr.some((t) => t.value === 'b')
                    ? arr.filter((t) => t.value !== 'b')
                    : [...arr, { value: 'b', label: 'LABEL' }],
                )
              }
            >
              Label B
            </DropdownItem>
          </DropdownMenu>
        }
      />
    </div>
  );
}

## Menu primitives

POD ships three menu primitives that pair with `Dropdown`:

- **`<DropdownMenu>`** — popover container (bg, border, shadow, scroll cap).
- **`<DropdownItem>`** — individual row. Supports `selected`, `disabled`, `destructive`, `error`, `leftAdornment`, `rightAdornment`, `showSelectedMark`.
- **`<DropdownBadge>`** — small leading badge with colored dot (9 colors). Drop as `leftAdornment`.

`Dropdown` itself remains stateless — you wire `open` + the menu yourself. Three composition examples below mirror the Figma source.

export function BooksMenuExample() {
  const [open, setOpen] = useState(true);
  const [selected, setSelected] = useState(['CIRC', 'NOVG']);
  const books = [
    { code: 'CIRC', name: 'Circa Sports', color: 'green' },

    { code: 'PINY', name: 'Pinnacle',     color: 'orange' },

    { code: 'BTNO', name: 'BetNow',       color: 'purple' },
    { code: 'UNDG', name: 'Underdog',     color: 'yellow' },

                      <DropdownBadge label={b.code} color={b.color} />
                    </span>
                  }
                  onClick={() => toggle(b.code)}
                >
                  {b.name}
                </DropdownItem>
              );
            })}
          </DropdownMenu>
        }
      />
    </div>
  );
}

## Props

| Prop | Type | Default | Notes |
|------|------|---------|-------|
| `label` | `string` | — | Visible label above. |
| `required` | `boolean` | `false` | Adds red asterisk to label. |
| `sublabel` | `string` | — | Secondary label text, e.g. `"(Optional)"`. |
| `labelInfo` | `boolean` | `false` | Info icon next to label. |
| `hint` | `string` | — | Hint text below, with info icon. |
| `error` | `string` | — | Replaces hint, applies destructive border + ring. |
| `size` | `'sm' \| 'md'` | `'md'` | Padding scale. |
| `variant` | `'default' \| 'tags'` | `'default'` | Trigger content style. |
| `selectedLabel` | `string` | — | For `default`: visible selected text. |
| `placeholder` | `string` | `'Select'` | Shown when not filled. |
| `tags` | `DropdownTag[]` | `[]` | For `tags`: list of `{ value, label }`. |
| `onRemoveTag` | `(value) => void` | — | Required to render X on chips. |
| `open` | `boolean` | `false` | Flips chevron + locks border. Consumer-controlled. |
| `disabled` | `boolean` | `false` | Standard disabled — hides chip X, disables click. |
| `onClick` | native button | — | Wire this to toggle your menu open/close. |

Extends all native `<button>` HTMLAttributes except `size` and `value`.

## Accessibility

- Native `<button>` trigger — keyboard activation (Space / Enter) works out of the box.
- `aria-haspopup="listbox"` + `aria-expanded={open}` so screen readers announce dropdown semantics.
- `aria-invalid` is set when `error` is present; error span is linked via `aria-describedby`.
- Hint span linked via `aria-describedby` when no error present.
- Chip X button is `tabIndex={-1}` to keep tab order on the trigger; remove via mouse only OR move focus management to consumer.
- Decorative icons (chevron, label info, chip X) are `aria-hidden`.

## Things to watch

- **Component is stateless.** It does NOT manage menu visibility or option lists. You wire that up — Dropdown only renders the trigger UI and flips the chevron based on `open`.
- **Chip click vs trigger click.** Clicking the X on a chip calls `onRemoveTag` and stops propagation, so the trigger's `onClick` (menu toggle) doesn't fire. Verify your menu doesn't open accidentally on chip remove.
- **Disabled hides X buttons too** — `onRemoveTag` won't render the X when `disabled`, even if the callback is passed. Re-enable to expose chip removal.
- **Truncation:** `default` variant uses `truncate` on the selected text — long selections get ellipsis. `tags` variant wraps to multiple lines (no truncation by default).
- **No multi-select API built in.** For `tags`, you maintain the array externally. POD doesn't ship a Combobox / multi-select primitive yet — confirm with designer if you need search-as-you-type inside the menu.
