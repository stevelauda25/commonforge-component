import { Combobox } from '@sparc-ui';
import { useState } from 'react';

export const projects = [
  { value: 'holly-hills', label: 'Holly Hills Repl ES' },
  { value: 'fairview', label: 'BVSD Fairview HS' },
  { value: 'creekside', label: 'Creekside Elem Addition' },
  { value: 'mapleton', label: 'Mapleton Admin Reno' },
  { value: 'svrain-ic', label: 'St Vrain Innovation Ctr' },
  { value: 'boulder-tec', label: 'Boulder TEC Wing C', disabled: true },
];

export function ComboboxDemo() {
  const [project, setProject] = useState('fairview');
  return (
    <div className="w-[460px]">
      <Combobox
        label="Project"
        required
        options={projects}
        value={project}
        onChange={setProject}
        placeholder="Select project"
        hint="Choose the active project for this estimate."
      />
    </div>
  );
}

<PageHeader

  description="Complete select-style form field — label row, dropdown, and hint row. Composes Dropdown."
/>
