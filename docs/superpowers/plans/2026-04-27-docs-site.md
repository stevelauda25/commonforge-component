# POD Docs Site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `apps/docs` — a Vite + React + MDX documentation site for CF Design System with hybrid kumo+asto layout, side-by-side with the existing `apps/playground`.

**Architecture:** New workspace package at `apps/docs`. Vite SPA with React Router (data router), MDX pipeline via `@mdx-js/rollup` + `remark-gfm` + `rehype-pretty-code`. Single source of truth for navigation/routes in `src/lib/routes.ts`. Light/dark theming via existing `@pod/tokens` CSS variables. Live component demos import directly from `@pod/ui`.

**Tech Stack:** Vite 5, React 18, React Router 6 (data router), Tailwind 3, `@mdx-js/rollup`, `remark-gfm`, `rehype-pretty-code` (Shiki), `lucide-react`, `@pod/ui`, `@pod/tokens`.

**Important conventions to follow:**
- ESM imports use `.js` extensions even for `.tsx` files (matches existing playground/UI convention).
- No hex colors in components/MDX content — consume `@pod/tokens` semantic tokens via Tailwind classes only.
- No automated tests — project explicitly defers tests (per `docs/DESIGN.md`). Each task ends with manual verification: typecheck pass + dev server browser check.
- This project is **not yet a git repo**. Commit steps are written assuming `git init` happens in Task 1. If the user explicitly opts out of git, skip the commit step at the end of each task — but still complete the verification.

- [ ] **Step 2: Create `apps/docs/src/lib/theme.ts`**

```ts
import { useEffect, useState } from 'react';

export type Theme = 'light' | 'dark';

const STORAGE_KEY = 'pod-docs-theme';

function readInitialTheme(): Theme {
  if (typeof window === 'undefined') return 'light';
  const stored = window.localStorage.getItem(STORAGE_KEY);
  if (stored === 'light' || stored === 'dark') return stored;
  return window.matchMedia('(prefers-color-scheme: dark)').matches
    ? 'dark'
    : 'light';
}

export function useTheme() {
  const [theme, setTheme] = useState<Theme>(readInitialTheme);

  useEffect(() => {
    const root = document.documentElement;
    root.classList.toggle('dark', theme === 'dark');
    window.localStorage.setItem(STORAGE_KEY, theme);
  }, [theme]);

  const toggle = () => setTheme((t) => (t === 'light' ? 'dark' : 'light'));

  return { theme, setTheme, toggle };
}
```

```tsx
import { cn } from '@pod/ui';
import {
  routes,
  componentRoutes,
  foundationRoutes,
  resourceRoutes,
} from '../../lib/routes.js';

const STORAGE_KEY = 'pod-docs-sidebar-groups';

function readGroupState(): Record<string, boolean> {
  if (typeof window === 'undefined') return {};
  try {
    return JSON.parse(window.localStorage.getItem(STORAGE_KEY) || '{}');
  } catch {
    return {};
  }
}

function writeGroupState(state: Record<string, boolean>) {
  window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
}

interface NavItemProps {
  to: string;
  label: string;
  status?: 'ready' | 'planned';
}

function NavItem({ to, label, status }: NavItemProps) {
  return (
    <NavLink
      to={to}
      className={({ isActive }) =>
        cn(
          'flex items-center gap-2 rounded-md px-3 py-1.5 text-sm transition-colors',
          'hover:bg-muted',
          isActive
            ? 'bg-muted text-text-primary font-medium'
            : 'text-text-secondary',
          status === 'planned' && 'opacity-60',
        )
      }
    >
      {status && (
        <span
          aria-hidden="true"
```

```tsx
import { useState } from 'react';
import { Check, Copy } from 'lucide-react';
import { cn } from '@pod/ui';

type Props = React.HTMLAttributes<HTMLPreElement> & {
  'data-language'?: string;
};

export function CodeBlock({ className, children, ...rest }: Props) {
  const [copied, setCopied] = useState(false);
  const language = (rest as Record<string, string>)['data-language'];

  const handleCopy = (event: React.MouseEvent<HTMLButtonElement>) => {
    const pre = event.currentTarget.closest('.code-block')?.querySelector('pre');
    const text = pre?.textContent ?? '';
    void navigator.clipboard.writeText(text).then(() => {
      setCopied(true);
      setTimeout(() => setCopied(false), 1200);
    });
  };

  return (
    <div className="code-block group relative my-4 overflow-hidden rounded-md border border-border-default bg-muted">
      {language && (
        <span className="absolute left-3 top-2 text-[10px] font-mono uppercase tracking-wider text-text-muted">
          {language}
        </span>
      )}
      <button
        type="button"
        onClick={handleCopy}
        aria-label="Copy code"
        className={cn(
          'absolute right-2 top-2 inline-flex h-7 w-7 items-center justify-center rounded',
          'text-text-muted opacity-0 transition-opacity hover:bg-canvas hover:text-text-primary',
          'group-hover:opacity-100 focus-visible:opacity-100',
        )}
      >
        {copied ? <Check className="h-3.5 w-3.5" /> : <Copy className="h-3.5 w-3.5" />}
      </button>
      <pre className={cn('overflow-x-auto p-4 pt-7 text-[13px] leading-snug', className)} {...rest}>
```

```tsx
import { MdxLayout } from '../components/docs/MdxLayout.js';

function isMdxPath(path: string) {
  return (
    path.startsWith('/components/') ||
    path.startsWith('/foundations/') ||
    path === '/getting-started' ||
    path === '/changelog'
  );
}

export const router = createBrowserRouter([
  {
    path: '/',
    Component: App,
    children: [
      ...routes.map((r) => ({
        path: r.path === '/' ? undefined : r.path.replace(/^\//, ''),
        index: r.path === '/',
        lazy: r.load
          ? async () => {
              const mod = await r.load!();
              const Inner = mod.default;
              const Wrapped = isMdxPath(r.path)
                ? () => (
                    <MdxLayout>
                      <Inner />
                    </MdxLayout>
                  )
                : Inner;
              return { Component: Wrapped };
            }
          : undefined,
      })),
      {
        path: '*',
        lazy: async () => {
          const mod = await import('../pages/NotFound.js');
```

## Architecture

Two layers of design tokens:

1. **Primitives** (`packages/tokens/src/primitives.ts`) — raw color scales (neutral, blue, green, red, …) stored as `"R G B"` triples for Tailwind alpha composition. Components never import these.
2. **Semantic** (`packages/tokens/src/theme.css`) — CSS custom properties defining the component vocabulary: `bg.canvas`, `text.primary`, `accent.*`, etc. Two scopes: `:root` (light) and `.dark`. **This is what components consume**, via the Tailwind preset.

The bridge is `packages/tokens/src/tailwind-preset.ts` — every semantic CSS variable is mapped to a Tailwind class (`bg-canvas`, `text-primary`, `border-border-default`, …). Both `@pod/ui` and the apps extend this preset.

## Component rules

- **TypeScript + React 18**, `forwardRef`, semicolons, plain exports.
- Classes composed via `cn()` (clsx + tailwind-merge) so consumers can override with `className` safely.
- Focus affordance is shared (`packages/ui/src/lib/focus-ring.ts`).
- **No hardcoded hex values** in `packages/ui/src/` — verifiable by grep.
- Tooltip uses Radix primitives for accessibility. Everything else is native HTML + Tailwind.

```tsx
          status === 'planned' && 'opacity-60',
        )
      }
    >
      {status && (
        <span
          aria-hidden="true"
          className={cn(
            'inline-block h-1.5 w-1.5 rounded-full',
            status === 'ready' ? 'bg-accent' : 'border border-text-muted',
          )}
        />
      )}
      <span className="truncate">{label}</span>
    </NavLink>
  );
}

interface GroupProps {
  id: string;
  label: string;
  children: React.ReactNode;
}

function Group({ id, label, children }: GroupProps) {
  const initial = readGroupState();
  const open = initial[id] ?? true;
  return (
    <details
      open={open}
      onToggle={(e) => {
        const state = readGroupState();
        state[id] = (e.currentTarget as HTMLDetailsElement).open;
        writeGroupState(state);
      }}
      className="group"
    >
      <summary className="flex cursor-pointer list-none items-center gap-1 px-3 py-1.5 text-xs font-semibold uppercase tracking-wide text-text-muted hover:text-text-secondary">
        <span className="inline-block h-3 w-3 transition-transform group-open:rotate-90">▸</span>
        {label}
      </summary>
      <div className="mt-1 flex flex-col gap-0.5">{children}</div>
    </details>
  );
}

interface SidebarProps {
  onNavigate?: () => void;
}

export function Sidebar({ onNavigate }: SidebarProps) {
  const top = routes.filter((r) => r.category === 'top');

  return (
    <aside className="flex h-full w-60 shrink-0 flex-col gap-2 border-r border-border-default bg-surface px-3 py-4">
      <nav className="flex flex-col gap-0.5">
        {top.map((r) => (
          <NavItem key={r.path} to={r.path} label={r.label} onNavigate={onNavigate} />
        ))}
      </nav>

      {foundationRoutes.length > 0 && (
        <Group id="foundations" label="Foundations">
          {foundationRoutes.map((r) => (
            <NavItem key={r.path} to={r.path} label={r.label} onNavigate={onNavigate} />
          ))}
        </Group>
      )}

      {componentRoutes.length > 0 && (
        <Group id="components" label="Components">
          {componentRoutes.map((r) => (
            <NavItem key={r.path} to={r.path} label={r.label} status={r.status} onNavigate={onNavigate} />
          ))}
        </Group>
      )}

      {resourceRoutes.length > 0 && (
        <Group id="resources" label="Resources">
          {resourceRoutes.map((r) => (
            <NavItem key={r.path} to={r.path} label={r.label} onNavigate={onNavigate} />
          ))}
        </Group>
      )}
    </aside>
  );
}
```

```tsx
import { useEffect } from 'react';
import { Sidebar } from './Sidebar.js';
import { cn } from '@pod/ui';

interface Props {
  open: boolean;
  onClose: () => void;
}

export function SidebarDrawer({ open, onClose }: Props) {
  useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };
    document.addEventListener('keydown', onKey);
    document.body.style.overflow = 'hidden';
    return () => {
      document.removeEventListener('keydown', onKey);
      document.body.style.overflow = '';
    };
  }, [open, onClose]);

  return (
    <>
      <div
        aria-hidden={!open}
        onClick={onClose}
        className={cn(
          'fixed inset-0 z-40 bg-overlay/40 transition-opacity md:hidden',
          open ? 'opacity-100 pointer-events-auto' : 'opacity-0 pointer-events-none',
        )}
      />
      <div
        role="dialog"
        aria-label="Navigation"
        aria-modal="true"
        className={cn(
          'fixed inset-y-0 left-0 z-50 w-60 transform transition-transform md:hidden',
          open ? 'translate-x-0' : '-translate-x-full',
        )}
      >
        <Sidebar onNavigate={onClose} />
      </div>
    </>
  );
}
```
